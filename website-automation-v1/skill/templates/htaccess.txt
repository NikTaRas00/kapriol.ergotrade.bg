# ============================================================================
# Apache config for shared cPanel-class hosting.  Copy to  .htaccess
# Adjust: the www/HTTPS canonical block, and any subfolder prefix on the
# ErrorDocument paths if the site is not at the domain root.
# ============================================================================

Options -Indexes -MultiViews
# index.html first (public pages), then index.php so a PHP-only directory like
# /admin/ still resolves its index.
DirectoryIndex index.html index.php

# -------------------------------------------------------------- Private data --
# Never serve admin secrets / logs, the shared guard include, or temp files.
<FilesMatch "(config|enquiries|login-throttle|analytics)\.(php|json)$">
  Require all denied
</FilesMatch>
<FilesMatch "(_guard\.php|\.tmp)$">
  Require all denied
</FilesMatch>
# NOTE: items.json / content.json / attributes.json are intentionally NOT blocked
# — the public site fetches them; they are only WRITABLE via the authed admin API.

# ---------------------------------------------------------------- Redirects --
<IfModule mod_rewrite.c>
  RewriteEngine On

  # Forbid the private/ data folder if it ever sits inside the web root.
  RewriteRule ^private/ - [F,L]

  # www -> non-www, forcing HTTPS in the same hop.
  RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
  RewriteRule ^ https://%1%{REQUEST_URI} [L,R=301]

  # Force HTTPS for the non-www host.
  RewriteCond %{HTTPS} off
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

  # Dynamic sitemap: /sitemap.xml is generated by sitemap.php from the live data.
  RewriteRule ^sitemap\.xml$ sitemap.php [L]

  # ---------------------------------------------- Clean (extensionless) URLs --
  # Uses %{THE_REQUEST} so it works at the web root AND inside a subfolder.
  # 1) ".../index.html" -> ".../"      (canonical home)
  RewriteCond %{THE_REQUEST} \s/+(([^\s?]*/)?)index\.html[\s?] [NC]
  RewriteRule ^ /%1 [R=301,L,NE]
  # 2) ".../page.html"   -> ".../page" (strip .html elsewhere; query kept)
  RewriteCond %{THE_REQUEST} \s/+([^\s?]+)\.html[\s?] [NC]
  RewriteRule ^ /%1 [R=301,L,NE]
  # 3) Internally serve "page" from "page.html" when that file exists.
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteCond %{REQUEST_FILENAME}\.html -f
  RewriteRule ^(.+?)/?$ $1.html [L]
</IfModule>

# ------------------------------------------------------------- Error pages --
# Self-contained pages in error/. Prefix these paths if the site moves into a
# subfolder (and update each error page's <base href>).
ErrorDocument 400 /error/400.html
ErrorDocument 401 /error/401.html
ErrorDocument 403 /error/403.html
ErrorDocument 404 /error/404.html
ErrorDocument 405 /error/405.html
ErrorDocument 429 /error/429.html
ErrorDocument 500 /error/500.html
ErrorDocument 502 /error/502.html
ErrorDocument 503 /error/503.html
ErrorDocument 504 /error/504.html

# ------------------------------------------------------------ Compression --
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css
  AddOutputFilterByType DEFLATE application/javascript application/x-javascript
  AddOutputFilterByType DEFLATE application/json image/svg+xml
</IfModule>

# --------------------------------------------------------------- Caching ----
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/css                "access plus 1 year"
  ExpiresByType application/javascript  "access plus 1 year"
  ExpiresByType application/x-javascript "access plus 1 year"
  ExpiresByType image/jpeg              "access plus 1 year"
  ExpiresByType image/png               "access plus 1 year"
  ExpiresByType image/webp              "access plus 1 year"
  ExpiresByType image/svg+xml           "access plus 1 year"
  ExpiresByType video/mp4               "access plus 1 year"
  ExpiresByType font/woff2              "access plus 1 year"
  ExpiresByType text/html               "access plus 0 seconds"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(css|js|jpg|jpeg|png|webp|svg|mp4|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
  # Admin-editable data must always revalidate so edits show immediately.
  <FilesMatch "\.json$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
  Header set X-Content-Type-Options "nosniff"
</IfModule>

# ----------------------------------------------------------- Mime fallback --
<IfModule mod_mime.c>
  AddType video/mp4 .mp4
  AddType image/svg+xml .svg
  AddType font/woff2 .woff2
</IfModule>
